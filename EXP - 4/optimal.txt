#include <iostream>
#include <vector>
using namespace std;

vector<int> rotateArray(vector<int> nums, int k) {
    int n = nums.size();
    k = k % n;

    vector<int> rotated;

    // Move elements from k to end
    for (int i = k; i < n; i++) {
        rotated.push_back(nums[i]);
    }

    // Move first k elements to end
    for (int i = 0; i < k; i++) {
        rotated.push_back(nums[i]);
    }

    return rotated;
}

int search(vector<int>& nums, int target) {
    int left = 0;
    int right = nums.size() - 1;

    while (left <= right) {
        int mid = (left + right) / 2;

        if (nums[mid] == target) {
            return mid;
        }

        // Left half is sorted
        if (nums[left] <= nums[mid]) {
            if (nums[left] <= target && target < nums[mid]) {
                right = mid - 1;
            }
            else {
                left = mid + 1;
            }
        }

        // Right half is sorted
        else {
            if (nums[mid] < target && target <= nums[right]) {
                left = mid + 1;
            }
            else {
                right = mid - 1;
            }
        }
    }

    return -1;
}

int main() {
    int n;

    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter the sorted array: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    int k;
    cout << "Enter rotation index: ";
    cin >> k;

    int target;
    cout << "Enter target element: ";
    cin >> target;

    vector<int> rotated = rotateArray(nums, k);

    cout << "Rotated Array: ";
    for (int i = 0; i < rotated.size(); i++) {
        cout << rotated[i] << " ";
    }

    int result = search(rotated, target);

    if (result != -1) {
        cout << "\nTarget found at index: " << result;
    }
    else {
        cout << "\nTarget not found.";
    }

    return 0;
}
