#include <iostream>
#include <vector>
using namespace std;

int searchInsert(vector<int>& nums, int target) {
    for (int i = 0; i < nums.size(); i++) {
        if (nums[i] >= target)
            return i;
    }
    return nums.size();
}

int main() {
    vector<int> nums = {2, 4, 6, 8, 10};
    int target = 7;

    cout << searchInsert(nums, target);
    return 0;
}
