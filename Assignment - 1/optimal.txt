#include <iostream>
#include <vector>
using namespace std;

bool search(vector<int> nums, int target) {
    for (int num : nums) {
        if (num == target) {
            return true;
        }
    }
    return false;
}

int main() {
    int n;
    
    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter the rotated sorted array: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    int target;
    cout << "Enter the target element: ";
    cin >> target;

    if (search(nums, target)) {
        cout << "True";
    } else {
        cout << "False";
    }

    return 0;
}
